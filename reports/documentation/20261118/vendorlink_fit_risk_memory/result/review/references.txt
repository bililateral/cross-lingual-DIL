def groups_for(accounts, seed, domain, epoch):
    ids = [a['account_uid'] for a in accounts]
    if ids != sorted(set(ids)) or len(ids) < 4:
        raise ValueError('Sorted unique fit accounts required')
    records = {}
    for a in accounts:
        items = sorted((Ad(**r) for r in a['records']), key=lambda r: r.uid)
        if len(items) != 12 or len({r.uid for r in items}) != 12:
            raise ValueError('Twelve distinct fit records required')
        random.Random(seed_for(seed, domain, epoch, 'records', a['account_uid'])).shuffle(items)
        records[a['account_uid']] = items
    result = []
    for packet in range(3):
        shuffled = ids.copy()
        random.Random(seed_for(seed, domain, epoch, 'accounts', packet)).shuffle(shuffled)
        for index, chunk in enumerate(group_chunks(shuffled, 8)):
            views = []
            for account in chunk:
                for v in range(2):
                    ads = tuple(records[account][4*packet+2*v:4*packet+2*v+2])
                    uid = hashlib.sha256(json.dumps([r.uid for r in ads]).encode()).hexdigest()
                    views.append(View(uid, ads, account))
            key = [seed, domain, epoch, packet, index, [[r.uid for r in v.records] for v in views]]
            uid = hashlib.sha256(json.dumps(key, ensure_ascii=False, separators=(',', ':')).encode()).hexdigest()
            g = Group(uid, domain, 'fit', tuple(views))
            g.validate(training=True)
            result.append(g)
    all_ids = [r.uid for g in result for v in g.views for r in v.records]
    if len(all_ids) != 12*len(accounts) or len(set(all_ids)) != len(all_ids):
        raise ValueError('Every current advertisement must appear once per epoch')
    return result

def group_chunks(values: list, identities_per_group: int) -> list[list]:
    chunks = [values[i:i+identities_per_group] for i in range(0, len(values), identities_per_group)]
    if chunks and len(chunks[-1]) < 4:
        if len(chunks) == 1:
            return []
        chunks[-2].extend(chunks.pop())
    return chunks

class Memory:
    def __init__(self, seed, settings):
        self.rng = random.Random(seed)
        self.settings = settings
        self.seen, self.entries = 0, []

    def retain(self, groups, stage, model, check=lambda: None, *, record_teacher=lambda group: None):
        if stage not in (1, 2):
            raise ValueError("Only completed nonfinal stages produce history")
        # Select first, then encode new survivors only; existing teachers never refresh.
        selected = list(self.entries)
        for group in sorted(groups, key=lambda g: g.uid):
            self.seen += 1
            slot = len(selected) if len(selected) < self.settings.memory_groups else self.rng.randrange(self.seen)
            if slot < self.settings.memory_groups:
                row = {"group": asdict(group), "origin": stage, "teacher": None}
                if slot == len(selected):
                    selected.append(row)
                else:
                    selected[slot] = row
        for row in selected:
            if row["teacher"] is None:
                group = Group.from_dict(row["group"])
                row["teacher"] = model.teacher(group, check)
                record_teacher(group)
        self.entries = selected
        if len(self.bytes()) > self.settings.memory_bytes:
            raise ValueError("Retained text, teachers and sampler state exceed the memory budget")

    def state(self):
        return {"seen": self.seen, "entries": self.entries, "rng": self.rng.getstate()}

    def restore(self, state):
        self.seen, self.entries = state["seen"], state["entries"]
        self.rng.setstate(state["rng"])

    def bytes(self):
        buffer = io.BytesIO()
        torch.save(self.state(), buffer)
        return buffer.getvalue()

    def draw(self, draw_seed):
        row = random.Random(draw_seed).choice(self.entries)
        return Group.from_dict(row["group"]), row["teacher"]

def read_endpoint(item, size):
    path, record, anchors = item
    data = path.read_bytes()
    require(len(data) == record["bytes"] and digest(data) == record["sha256"], "Matrix identity mismatch")
    scores = np.load(io.BytesIO(data), allow_pickle=False)
    require(scores.shape == (size, size) and scores.dtype.kind == "f" and np.isfinite(scores).all(),
            "Invalid score matrix")
    scores = scores.astype(np.float64, copy=False)
    truth = np.arange(size)
    order = np.argsort(-scores, axis=1, kind="stable")
    ranks = (order == truth[:, None]).argmax(axis=1) + 1
    winners = order[:, 0]
    hits = (winners == truth)
    predicted = np.bincount(winners, minlength=size)
    metrics = {"Top1": float(hits.mean()), "MAP": float((1.0 / ranks).mean()),
               "macro_F1": float((2.0 * hits / (1 + predicted)).mean())}
    require(set(metrics) == set(anchors) and all(abs(v - anchors[k]) <= 1e-12 for k, v in metrics.items()),
            "Saved endpoint metric or stable ordering mismatch")
    return scores, ranks

        packet = i // per_packet

            v0, v1 = g.views[k:k+2]
            require(v0.identity == v1.identity and len(v0.records) == len(v1.records) == 2,
                    "Original paired fit views required")
            account = id_to_index[v0.identity]
            members.append(account)
            for offset, v in enumerate((v0, v1)):
                require(views[account][2*packet+offset] is None, "Repeated fit view")
                views[account][2*packet+offset] = v.records
                view_records.append([v.uid, [r.uid for r in v.records]])
        require(len(set(members)) == len(members) and g.uid not in catalog, "Repeated group/account")
        n_records = 4 * len(members)

        slot = Slot(g.uid, stage, packet, tuple(members),
                    dict(group=asdict(g), origin=stage, teacher=teacher))
        pool.append(slot)
        catalog[g.uid] = dict(stage=stage, packet=packet, members=members, views=view_records)
    for account in views:
        require(all(v is not None for v in account) and len({r.uid for v in account for r in v}) == 12,
                "Six disjoint two-record fit views required")
    return pool, views, catalog


def cosine_field(references, queries):
    vec = TfidfVectorizer(lowercase=True, ngram_range=(1, 1), norm="l2", smooth_idf=True,
                          sublinear_tf=False, dtype=np.float64)
    if not any(vec.build_analyzer()(s) for s in references):
        return np.zeros((len(queries), len(references)))
    ref = vec.fit_transform(references)
    return (vec.transform(queries) @ ref.T).toarray()

def random_slots(old, ordered, rng, seen):
    result = list(old)
    for group in ordered:
        seen += 1
        slot = len(result) if len(result) < 6 else rng.randrange(seen)
        if slot < 6:
            if slot == len(result):
                result.append(group)
            else:
                result[slot] = group
    require(len(result) == 6, "Reservoir not full")
    return result, seen

def events(start, end):
    _, first_ranks = start
    scores, end_ranks = end
    winners = scores.argmax(axis=1)
    require(np.array_equal(winners == np.arange(len(winners)), end_ranks == 1), "Final tie order")
    lost = np.flatnonzero((first_ranks == 1) & (end_ranks != 1))
    return [(int(q), int(winners[q])) for q in lost]

def view_table(pool, n):
    table = [[None] * n for _ in range(3)]
    for slot in pool:
        views = slot.row["group"]["views"]
        for ix, member in enumerate(slot.members):
            require(table[slot.packet][member] is None, "Repeated template position")
            table[slot.packet][member] = views[2*ix:2*ix+2]
    require(all(all(packet) for packet in table), "Missing template position")
    return table

def materialize(pool, table, index, layout, seed):
    template = pool[index]
    group = template.row["group"]
    members = tuple(layout[p] for p in template.members)
    views = tuple(v for a in members for v in table[template.packet][a])
    records = [[r["uid"] for r in v["records"]] for v in views]
    uid = group_uid(seed, group["domain"], template.packet, index % (len(pool)//3), records)
    # Preserve the original Memory serializer's container types and distinct teacher storage.
    row = dict(group=dict(uid=uid, domain=group["domain"], split=group["split"], views=views),
               origin=template.stage, teacher=torch.zeros_like(template.row["teacher"]))
    return Slot(uid, template.stage, template.packet, members, row)
